cmake_minimum_required(VERSION 3.22)
project(Combination LANGUAGES C CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)

# Reuse the local RI dependency checkout without duplicating its vendor tree.
# The application/control backend is not yet migrated into this build.
set(COMBINATION_REFERENCE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../RecurseIntall/Crocker-Digitlization-GUI/CrockerGUI" CACHE PATH "RI dependency checkout")
set(vendor "${COMBINATION_REFERENCE_ROOT}/external")
find_package(OpenGL REQUIRED)
file(GLOB_RECURSE engine_sources CONFIGURE_DEPENDS source/Engine/*.cpp)
add_library(combination_graphics STATIC ${engine_sources} "${vendor}/glad/src/glad.c")
set_target_properties(combination_graphics PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(combination_graphics PUBLIC include "${vendor}/glad/include" "${vendor}/glm" "${vendor}")
target_link_libraries(combination_graphics PUBLIC OpenGL::GL)
target_compile_definitions(combination_graphics PRIVATE CROCKER_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets/engine")
if(MSVC)
    target_compile_options(combination_graphics PRIVATE /W4 /permissive-)
endif()
include(CTest)
add_library(combination_backend STATIC source/Simulation/SimulationRuntime.cpp source/Controls/ControlSystem/NLAPID.cpp)
set_target_properties(combination_backend PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(combination_backend PUBLIC include)
target_link_libraries(combination_backend PUBLIC Threads::Threads)
if(BUILD_TESTING)
    foreach(test TextLayoutTest GraphicsBufferTest CstTrendModelTest)
        add_executable(${test} tests/native/${test}.cpp)
        target_link_libraries(${test} PRIVATE combination_graphics)
        add_test(NAME ${test} COMMAND ${test})
    endforeach()
endif()
option(COMBINATION_BUILD_PYTHON "Build the native CST graph module" ON)
if(COMBINATION_BUILD_PYTHON)
    find_package(Python 3.13 REQUIRED COMPONENTS Interpreter Development.Module)
    execute_process(COMMAND "${Python_EXECUTABLE}" -m pybind11 --cmakedir
        OUTPUT_VARIABLE pybind11_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    find_package(pybind11 CONFIG REQUIRED)
    pybind11_add_module(CombinationCharts source/Bindings/ChartBindings.cpp)
    target_link_libraries(CombinationCharts PRIVATE combination_graphics)
    pybind11_add_module(CombinationBackend source/Bindings/SimulationBindings.cpp)
    target_link_libraries(CombinationBackend PRIVATE combination_backend)
endif()
